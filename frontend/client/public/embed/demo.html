<role-room-infographic> — embed hvor som helst

Rammeverk-agnostisk custom element. Malen leser window.__CFG__ (data) + definerer setProgress(p) (animasjon). Rendres isolert i en iframe.

Levende eksempel (mal via template-url, data via attributt)

Bruk

<script src="/embed/role-room-infographic.js"></script>

<!-- A) data via attributt, mal via template-url -->
<role-room-infographic accent="#2f6df0" autoplay="5"
  template-url="/maler/kpi.html"
  data='{"cards":[{"label":"Pasienter","value":"124","icon":"group"}]}'>
</role-room-infographic>

<!-- B) mal + data via JS-property (best for store objekter / rammeverk-binding) -->
<script>
  const el = document.querySelector('role-room-infographic');
  el.template = '<div id="wrap">...</div><script>window.setProgress=...</script>';
  el.data = { cards: [ /* ... */ ] };
  el.fontsCss = '@font-face{...}'; // valgfritt: ikon-/merkevare-fonter
</script>
